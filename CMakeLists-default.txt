cmake_minimum_required(VERSION 3.29)
project(cpp_app LANGUAGES CXX)

file(GLOB SOURCES CONFIGURE_DEPENDS *.cpp)

add_executable(cpp_app ${SOURCES})

set_target_properties(cpp_app PROPERTIES
    CXX_STANDARD 23
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
)

enable_testing()

add_test(NAME run_app COMMAND cpp_app)
